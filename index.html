import turtle # 1. Створення вікна гри wn = turtle.Screen() wn.title("Пін-Понг на Python") wn.bgcolor("black") wn.setup(width=800, height=600) wn.tracer(0) # Зупиняє оновлення екрана для плавного руху # Рахунок score_a = 0 score_b = 0 # 2. Ракетка А (ліва) paddle_a = turtle.Turtle() paddle_a.speed(0) paddle_a.shape("square") paddle_a.color("white") paddle_a.shapesize(stretch_wid=5, stretch_len=1) # Робимо прямокутник paddle_a.penup() paddle_a.goto(-350, 0) # 3. Ракетка Б (права) paddle_b = turtle.Turtle() paddle_b.speed(0) paddle_b.shape("square") paddle_b.color("white") paddle_b.shapesize(stretch_wid=5, stretch_len=1) paddle_b.penup() paddle_b.goto(350, 0) # 4. М'яч ball = turtle.Turtle() ball.speed(0) ball.shape("square") ball.color("red") ball.penup() ball.goto(0, 0) ball.dx = 0.15 # Швидкість руху по X ball.dy = 0.15 # Швидкість руху по Y # 5. Табло рахунку pen = turtle.Turtle() pen.speed(0) pen.color("white") pen.penup() pen.hideturtle() pen.goto(0, 260) pen.write("Гравець A: 0 | Гравець Б: 0", align="center", font=("Courier", 20, "normal")) # --- Функції руху ракеток --- def paddle_a_up(): y = paddle_a.ycor() if y < 240: # Обмеження екрана paddle_a.sety(y + 20) def paddle_a_down(): y = paddle_a.ycor() if y > -240: paddle_a.sety(y - 20) def paddle_b_up(): y = paddle_b.ycor() if y < 240: paddle_b.sety(y + 20) def paddle_b_down(): y = paddle_b.ycor() if y > -240: paddle_b.sety(y - 20) # Прив'язка клавіш управління wn.listen() # Управління лівою ракеткою (Клавіші W та S) wn.onkeypress(paddle_a_up, "w") wn.onkeypress(paddle_a_down, "s") # Управління правою ракеткою (Стрілки Вгору та Вниз) wn.onkeypress(paddle_b_up, "Up") wn.onkeypress(paddle_b_down, "Down") # --- Головний ігровий цикл --- while True: wn.update() # Рух м'яча ball.setx(ball.xcor() + ball.dx) ball.sety(ball.ycor() + ball.dy) # Перевірка меж екрана (Зверху / Знизу) if ball.ycor() > 290: ball.sety(290) ball.dy *= -1 # Зміна напрямку по Y if ball.ycor() < -290: ball.sety(-290) ball.dy *= -1 # Вихід м'яча за праву межу (Гол для Гравця А) if ball.xcor() > 390: ball.goto(0, 0) ball.dx *= -1 score_a += 1 pen.clear() pen.write(f"Гравець A: {score_a} | Гравець Б: {score_b}", align="center", font=("Courier", 20, "normal")) # Вихід м'яча за ліву межу (Гол для Гравця Б) if ball.xcor() < -390: ball.goto(0, 0) ball.dx *= -1 score_b += 1 pen.clear() pen.write(f"Гравець A: {score_a} | Гравець Б: {score_b}", align="center", font=("Courier", 20, "normal")) # Зіткнення м'яча з правою ракеткою if (ball.dx > 0) and (350 > ball.xcor() > 340) and (paddle_b.ycor() + 50 > ball.ycor() > paddle_b.ycor() - 50): ball.setx(340) ball.dx *= -1 # Зіткнення м'яча з лівою ракеткою if (ball.dx < 0) and (-350 < ball.xcor() < -340) and (paddle_a.ycor() + 50 > ball.ycor() > paddle_a.ycor() - 50): ball.setx(-340) ball.dx *= -1
